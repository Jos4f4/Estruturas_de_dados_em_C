// função básica: troca
void troca(int *x, int *y){
	int aux;
	aux = *x;
	*x = *y;
	*y = aux;
}

void selection(int vet[]){
	int i, j, min;
	for(i=0; i<N-1;i++){
		min=i;
		for(j=i;j<N;j++){
			if(vet[j]<vet[min]){
				min=j;
			}
		}
		troca(&vet[min], &vet[i]);
	}
}
