// função básica: troca
void troca(int *x, int *y){
	int aux;
	aux = *x;
	*x = *y;
	*y = aux;
}

void bubble(int vet[]){
	int i, j;
	for(i=0; i<N; i++){
		for(j=1; j<N; j++){
			if(vet[j-1]>vet[j]){
				troca(&vet[j], &vet[j-1]);
			}
		}
	}
}

void bubble(int vet[]){
	int i, j;
	for(i=0; i<N-1; i++){
		for(j=1; j<N-i; j++){

				troca(&vet[j], &vet[j-1]);
			}
		}
	}
}

//otimizado para evitar percorrer o vetor em algo já ordenado
